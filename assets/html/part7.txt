  function buildFeedback() {
    const results = window.simulationResults || {};
    const firstChoice = Number(results.firstInteractionChoice || 0);

    const greetingOk = firstChoice === 1 || firstChoice === 3;
    const formalityOk = firstChoice === 1 || firstChoice === 2;

    let greetingExplanation = '';
    let formalityExplanation = '';
    let solutionExplanation = '';

    const secondChoice = Number(results.secondInteractionChoice || 0);

    if (firstChoice === 2 || firstChoice === 4) {
      greetingExplanation =
        'A saudação ficou incompleta. Como o cliente iniciou o contato com “Boa tarde”, o atendente também deve responder ao período do dia, mantendo uma abertura cordial e adequada.';
    }

    if (firstChoice === 3) {
      formalityExplanation =
        'Evite tratamentos como “senhor”, “senhora” ou “meu caro”. No atendimento via chat, prefira “você”, mantendo uma comunicação profissional, próxima e natural.';
    }

    if (firstChoice === 4) {
      formalityExplanation =
        'Evite gírias e expressões excessivamente informais, como “beleza” e “meu caro”. Prefira abordagens como “Tudo bem?” ou “Tudo certo?” e utilize “você” ao se dirigir ao cliente.';
    }

    if (secondChoice === 1) {
      solutionExplanation =
        'Não direcione o cliente para o suporte da loja virtual antes de analisar o caso e confirmar que a situação está realmente fora do escopo do Bling. Esse encaminhamento só deve ser feito quando houver orientação expressa da liderança técnica de que o assunto não é de nossa responsabilidade.';
    }

    if (secondChoice === 3) {
      solutionExplanation =
        'Sempre conduza o cliente com orientações claras e objetivas para a resolução do problema. Apenas pedir que ele pesquise na Central de Ajuda transfere a busca pela solução ao cliente e demonstra falta de proatividade no atendimento.';
    }

    const criteria = [
      {
        name:'Saudação',
        ok:greetingOk,
        explanation:greetingExplanation
      },
      {
        name:'Formalidade',
        ok:formalityOk,
        explanation:formalityExplanation
      },
      {
        name:'Fornecimento da solução correta',
        ok:results.secondInteractionCorrect === true,
        explanation:solutionExplanation
      },
      {
        name:'Oferta de ajuda adicional',
        ok:results.thirdInteractionCorrect === true,
        explanation:''
      },
      {
        name:'Deixar à disposição',
        ok:results.fourthInteractionCorrect === true,
        explanation:''
      }
    ];

    const score = criteria.reduce((total, item) => total + (item.ok ? 20 : 0), 0);

    const scoreEl = document.getElementById('feedbackScore');
    const listEl = document.getElementById('feedbackList');

    if (scoreEl) {
      scoreEl.textContent = score;

      const scoreBox = scoreEl.closest('.feedback-score');
      if (scoreBox) {
        scoreBox.classList.remove(
          'score-100',
          'score-80',
          'score-60',
          'score-40',
          'score-20',
          'score-0'
        );
        scoreBox.classList.add('score-' + score);
      }
    }

    if (listEl) {
      listEl.innerHTML = criteria.map(item => `
        <div class="feedback-row ${!item.ok ? 'feedback-warning' : ''}">
          <div class="feedback-copy">
            <div class="feedback-name">${item.name}</div>
            ${!item.ok && item.explanation
              ? `<div class="feedback-explanation">${item.explanation}</div>`
              : ''}
          </div>
          <div class="feedback-result">
            <span class="feedback-status ${item.ok ? 'ok' : 'no'}">
              ${item.ok ? 'Atendeu' : 'Não atendeu'}
            </span>
            <span class="feedback-points">${item.ok ? '20' : '0'}</span>
          </div>
        </div>
      `).join('');
    }

    return score;
  }

  function showFeedback() {
    buildFeedback();
    const overlay = document.getElementById('feedbackOverlay');
    if (overlay) {
      overlay.classList.add('active');
      overlay.setAttribute('aria-hidden', 'false');
    }
  }

  function hideFeedback() {
    const overlay = document.getElementById('feedbackOverlay');
    if (overlay) {
      overlay.classList.remove('active');
      overlay.setAttribute('aria-hidden', 'true');
    }
  }
