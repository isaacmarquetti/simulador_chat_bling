  function positionTour() {
    if (!tourLayer.classList.contains('active')) return;
    const step = tourSteps[tourIndex];
    const target = document.querySelector(step.selector);
    if (!target) return;

    const r = target.getBoundingClientRect();
    const pad = 5;

    tourTitle.textContent = step.title;
    tourText.innerHTML = step.html || step.text;
    if (!step.center) tourTooltip.style.transform = 'none';
    tourProgress.textContent = (tourIndex + 1) + ' de ' + tourSteps.length;
    tourNext.textContent = tourIndex === tourSteps.length - 1 ? 'Concluir tutorial' : 'Prosseguir';
    tourSkip.style.display = step.hideSkip ? 'none' : 'inline-block';

    if (step.center) {
      tourTooltip.classList.add('centered');

      if (step.noHighlight) {
        tourHighlight.style.display = 'block';
        tourHighlight.style.left = '0px';
        tourHighlight.style.top = '0px';
        tourHighlight.style.width = '100vw';
        tourHighlight.style.height = '100vh';
        tourHighlight.style.border = '0';
        tourHighlight.style.borderRadius = '0';
        tourHighlight.style.background = 'rgba(15,23,42,.58)';
        tourHighlight.style.boxShadow = 'none';
      } else {
        tourHighlight.style.display = 'block';
        tourHighlight.style.border = '3px solid #22c55e';
        tourHighlight.style.borderRadius = '10px';
        tourHighlight.style.background = 'transparent';
        tourHighlight.style.boxShadow = '0 0 0 9999px rgba(15,23,42,.58)';
        tourHighlight.style.left = (r.left - pad) + 'px';
        tourHighlight.style.top = (r.top - pad) + 'px';
        tourHighlight.style.width = (r.width + pad * 2) + 'px';
        tourHighlight.style.height = (r.height + pad * 2) + 'px';
      }

      tourTooltip.style.left = '50%';
      tourTooltip.style.top = '50%';
      tourTooltip.style.transform = 'translate(-50%, -50%)';
      return;
    }

    tourTooltip.classList.remove('centered');
    tourTooltip.style.transform = 'none';
    tourHighlight.style.display = 'block';
    tourHighlight.style.border = '3px solid #22c55e';
    tourHighlight.style.borderRadius = '10px';
    tourHighlight.style.background = 'transparent';
    tourHighlight.style.boxShadow = '0 0 0 9999px rgba(15,23,42,.58)';
    tourHighlight.style.left = (r.left - pad) + 'px';
    tourHighlight.style.top = (r.top - pad) + 'px';
    tourHighlight.style.width = (r.width + pad * 2) + 'px';
    tourHighlight.style.height = (r.height + pad * 2) + 'px';

    // Decide automaticamente de que lado o balão cabe melhor.
    const tooltipWidth = Math.min(360, window.innerWidth - 32);
    const tooltipHeight = 210;
    const gap = 14;

    let left = r.right + gap;
    let top = r.top + 12;

    if (left + tooltipWidth > window.innerWidth - 12) {
      left = r.left - tooltipWidth - gap;
    }
    if (left < 12) {
      left = Math.max(12, Math.min(window.innerWidth - tooltipWidth - 12, r.left + 12));
      top = r.bottom + gap;
    }
    if (top + tooltipHeight > window.innerHeight - 12) {
      top = Math.max(12, window.innerHeight - tooltipHeight - 12);
    }

    tourTooltip.style.left = left + 'px';
    tourTooltip.style.top = top + 'px';
  }

  function updateConversationPreview(sender, content, isHtml = true) {
    const preview = document.getElementById('conversationPreview');
    if (!preview) return;

    let text = content;

    if (isHtml) {
      const temp = document.createElement('div');
      temp.innerHTML = content;
      text = temp.textContent || temp.innerText || '';
    }

    text = String(text).replace(/\s+/g, ' ').trim();

    // No Octa, mensagens enviadas pelo atendente aparecem com o indicador de leitura.
    if (sender === 'agent') {
      preview.innerHTML = '<span class="double-check">✓✓</span>' + text;
    } else {
      preview.textContent = text;
    }

    const previewTime = document.getElementById('conversationPreviewTime');
    if (previewTime) {
      previewTime.textContent = formatTime(new Date());
    }
  }
