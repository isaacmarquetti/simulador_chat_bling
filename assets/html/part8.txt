        window.simulationResults.secondInteractionAnswer = button.innerText.trim();

        addChatMessage('agent', chosenText);

        const currentSecondBox = document.getElementById('secondInteractionBox');
        if (currentSecondBox) currentSecondBox.style.display = 'none';

        // Dá tempo para a conversa respirar antes da reação do cliente.
        clearTimeout(secondClientTypingTimer);
        clearTimeout(secondClientReplyTimer);
        clearTimeout(finalClientTypingTimer);
        clearTimeout(finalClientReplyTimer);

        if (clientOutcome === 'shopee') {
          secondClientTypingTimer = setTimeout(() => {
            addTypingIndicator('Cristiano Ronaldo');
            secondClientTypingTimer = null;
          }, 900);

          secondClientReplyTimer = setTimeout(() => {
            removeTypingIndicator();
            addChatMessage('client', 'Que pena...');
            secondClientReplyTimer = null;

            extraClientMessageTimer = setTimeout(() => {
              addChatMessage('client', 'Vou entrar em contato com o suporte da Shopee.');

              extraClientMessageTimer = setTimeout(() => {
                addChatMessage('client', 'Obrigado');
                extraClientMessageTimer = null;

                // Depois da reação do cliente, o atendimento segue para a finalização.
                addThirdInteraction();
              }, 1200);
            }, 1200);
          }, 2400);
        } else {
          secondClientTypingTimer = setTimeout(() => {
            addTypingIndicator('Cristiano Ronaldo');
            secondClientTypingTimer = null;
          }, 900);

          secondClientReplyTimer = setTimeout(() => {
            removeTypingIndicator();
            addChatMessage('client', 'Ok, vou verificar.');
            secondClientReplyTimer = null;

            // Aguarda 5 segundos antes da confirmação do cliente.
            finalClientTypingTimer = setTimeout(() => {
              addTypingIndicator('Cristiano Ronaldo');

            finalClientReplyTimer = setTimeout(() => {
              removeTypingIndicator();

              finalClientReplyTimer = null;

              extraClientMessageTimer = setTimeout(() => {

                extraClientMessageTimer = null;

                addThirdInteraction();
              }, 1200);
            }, 5000);
          }, 2400);
        }

      if (interaction === '3') {
        if (thirdAnswerSelected) return;
        thirdAnswerSelected = true;

        const clientOutcome = button.getAttribute('data-client-outcome');

          }

          } else if (clientOutcome === 'continue-email') {
            addChatMessage('client', 'Perfeito, anotado!');
          } else {
            addChatMessage('client', 'nada mais');

          addFourthInteraction();

      if (interaction === '4') {
        if (fourthAnswerSelected) return;
        fourthAnswerSelected = true;

  const openList = document.getElementById('openList');
  const emptyList = document.getElementById('emptyList');

  // "Encerradas" e "Nova conversa" são apenas elementos visuais neste treinamento.
  // A aba "Abertas" permanece fixa durante toda a simulação.
  tabs.forEach(t => {
    if (t.dataset.tab !== 'open') {
      t.setAttribute('aria-disabled', 'true');
      t.tabIndex = -1;
    }
  });

  const form = document.getElementById('composerForm');
  const input = document.getElementById('composer');
  const messages = document.getElementById('messages');

  // O campo de resposta do Octa é apenas visual.
  if (form) {
    form.addEventListener('submit', e => e.preventDefault());
  }
  if (input) {
    input.addEventListener('keydown', e => e.preventDefault());
    input.addEventListener('paste', e => e.preventDefault());
    input.addEventListener('drop', e => e.preventDefault());
  }


  document.addEventListener('click', (event) => {
    const closeFeedbackButton = event.target.closest('#feedbackClose');
    if (closeFeedbackButton) {
      hideFeedback();
      return;
    }

    const restartFeedbackButton = event.target.closest('#feedbackRestart');
    if (restartFeedbackButton) {
      hideFeedback();

      // Reinicia usando exatamente a mesma rotina do botão superior.
      const topRestart = document.getElementById('restartButton');
      if (topRestart) {
        topRestart.click();
      }
      return;
    }
  });

})();
