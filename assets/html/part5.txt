<script>
(function(){
  const startButton = document.getElementById('startSimulation');
  const homeButton = document.getElementById('homeButton');
  const tutorialButton = document.getElementById('tutorialButton');
  const restartButton = document.getElementById('restartButton');
  const helpPopover = document.getElementById('helpPopover');
  const helpPopoverTitle = document.getElementById('helpPopoverTitle');
  const helpPopoverText = document.getElementById('helpPopoverText');
  const helpPopoverClose = document.getElementById('helpPopoverClose');
  let contextualHelpUnlocked = false;

  function pad2(value) {
    return String(value).padStart(2, '0');
  }

  function formatDateTime(date) {
    return pad2(date.getDate()) + '/' +
           pad2(date.getMonth() + 1) + '/' +
           date.getFullYear() + ' às ' +
           pad2(date.getHours()) + ':' +
           pad2(date.getMinutes());
  }

  function formatTime(date) {
    return pad2(date.getHours()) + ':' + pad2(date.getMinutes());
  }

  function updateSimulationTimes() {
    const startedAt = new Date();

    const dauStartedAt = new Date(startedAt.getTime() - (7 * 60 * 1000));
    const firstMessageAt = new Date(startedAt.getTime() - (1 * 60 * 1000));

    const dauWindowText = document.getElementById('dauWindowText');
    const initialMessageTime = document.getElementById('initialMessageTime');
    const conversationPreviewTime = document.getElementById('conversationPreviewTime');

    if (dauWindowText) {
      dauWindowText.textContent =
        'Janela de cobrança DAU iniciada em ' +
        formatDateTime(dauStartedAt) +
        '⌄';
    }

    if (initialMessageTime) {
      initialMessageTime.textContent = formatTime(firstMessageAt);
    }

    if (conversationPreviewTime) {
      conversationPreviewTime.textContent = formatTime(firstMessageAt);
    }
  }

  function hideContextHelp() {
    if (helpPopover) helpPopover.classList.remove('active');
  }

  function showContextHelp(target) {
    if (!contextualHelpUnlocked || !target || !helpPopover) return;
    const title = target.getAttribute('data-help-title');
    const text = target.getAttribute('data-help-text');
    if (!title || !text) return;

    helpPopoverTitle.textContent = title;
    helpPopoverText.textContent = text;
    helpPopover.classList.add('active');
    helpPopover.classList.remove('placement-right', 'placement-left', 'placement-bottom', 'placement-top');

    const rect = target.getBoundingClientRect();
    const gap = 12;
    const width = Math.min(320, window.innerWidth - 24);
    const estimatedHeight = 170;

    const fitsRight = rect.right + gap + width <= window.innerWidth - 12;
    const fitsLeft = rect.left - gap - width >= 12;
    const fitsBottom = rect.bottom + gap + estimatedHeight <= window.innerHeight - 12;

    let left = 12;
    let top = 12;
    let placement = 'right';

    if (fitsRight) {
      placement = 'right';
      left = rect.right + gap;
      top = rect.top + (rect.height / 2) - (estimatedHeight / 2);
    } else if (fitsLeft) {
      placement = 'left';
      left = rect.left - width - gap;
      top = rect.top + (rect.height / 2) - (estimatedHeight / 2);
    } else if (fitsBottom) {
      placement = 'bottom';
      left = rect.left + (rect.width / 2) - (width / 2);
      top = rect.bottom + gap;
    } else {
      placement = 'top';
      left = rect.left + (rect.width / 2) - (width / 2);
      top = rect.top - estimatedHeight - gap;
    }

    if (left + width > window.innerWidth - 12) left = window.innerWidth - width - 12;
    if (left < 12) left = 12;
    if (top + estimatedHeight > window.innerHeight - 12) top = window.innerHeight - estimatedHeight - 12;
    if (top < 12) top = 12;

    helpPopover.style.left = left + 'px';
    helpPopover.style.top = top + 'px';
    helpPopover.classList.add('placement-' + placement);

    if (placement === 'right' || placement === 'left') {
      const arrowTop = Math.max(16, Math.min(estimatedHeight - 28, rect.top - top + (rect.height / 2)));
      helpPopover.style.setProperty('--arrow-top', arrowTop + 'px');
    } else {
      const arrowLeft = Math.max(18, Math.min(width - 28, rect.left - left + (rect.width / 2)));
      helpPopover.style.setProperty('--arrow-left', arrowLeft + 'px');
    }
  }

  if (startButton) {
    startButton.addEventListener('click', () => {
      updateSimulationTimes();
      document.body.classList.add('simulation-started');
      window.scrollTo(0, 0);
      setTimeout(startTour, 250);
    });
  }

  if (homeButton) {
    homeButton.addEventListener('click', () => {
      endTour();
      hideContextHelp();
      document.body.classList.remove('simulation-started');
      window.scrollTo(0, 0);
    });
  }

  if (tutorialButton) {
    tutorialButton.addEventListener('click', () => {
      hideContextHelp();
      startTour();
    });
  }

  if (restartButton) {
    restartButton.addEventListener('click', () => {
      hideFeedback();
      hideContextHelp();
      if (tourLayer) tourLayer.classList.remove('active');

      // Cancela qualquer ação pendente do fluxo anterior.
      clearTimeout(firstReplyTimer);
      clearTimeout(firstRevealTimer);
      clearTimeout(secondRevealTimer);
      clearTimeout(typingTimer);
      clearTimeout(secondClientReplyTimer);
      clearTimeout(secondClientTypingTimer);
      clearTimeout(finalClientReplyTimer);
      clearTimeout(finalClientTypingTimer);
      clearTimeout(extraClientMessageTimer);
      clearTimeout(thirdRevealTimer);
      clearTimeout(thirdClientTypingTimer);
      clearTimeout(thirdClientReplyTimer);
      clearTimeout(fourthRevealTimer);

      firstReplyTimer = null;
      firstRevealTimer = null;
      secondRevealTimer = null;
      typingTimer = null;
      secondClientReplyTimer = null;
      secondClientTypingTimer = null;
      finalClientReplyTimer = null;
      finalClientTypingTimer = null;
      extraClientMessageTimer = null;
      thirdRevealTimer = null;
      thirdClientTypingTimer = null;
      thirdClientReplyTimer = null;
      fourthRevealTimer = null;

      removeTypingIndicator();

      const messages = document.getElementById('messages');
      const composerInput = document.getElementById('composer');

      // Restaura exatamente o estado inicial da conversa.
      if (messages && window.initialMessagesHTML) {
        messages.innerHTML = window.initialMessagesHTML;
        messages.scrollTop = 0;
      }
